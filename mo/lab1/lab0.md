import math
import matplotlib.pyplot as plt

def target_function(x):
  #return (x - 2)**2
  x_ = x/2.
  return 0.5 - (x_) * math.exp(-(x_**2))

START_A = 0.
START_B = 5.
VARIANT_EPS = 1e-3

def method_dichotomy(func, a, b, eps, verbose=False):
    delta = eps / 2.1  
    calls = 0
    history = []
    iteration = 0
    prev_len = b - a
    
    while (b - a) > eps:
        iteration += 1
        x1 = (a + b - delta) / 2
        x2 = (a + b + delta) / 2
        
        fx1 = func(x1)
        fx2 = func(x2)
        calls += 2
        
        current_len = b - a
        ratio = prev_len / current_len if iteration > 1 else 0
        
        if verbose:
            history.append({
                'i': iteration,
                'x1': x1, 'x2': x2,
                'fx1': fx1, 'fx2': fx2,
                'a': a, 'b': b,
                'len': current_len,
                'ratio': ratio
            })
            
        prev_len = current_len

        if fx1 < fx2:
            b = x2
        else:
            a = x1
            
    min_x = (a + b) / 2
    return min_x, func(min_x), calls, history


def method_golden_section(func, a, b, eps, verbose=False):
    phi = (1 + math.sqrt(5)) / 2 # Золотая пропорция
    calls = 0
    history = []
    iteration = 0
    
    # Инициализация точек
    x1 = b - (b - a) / phi
    x2 = a + (b - a) / phi
    fx1 = func(x1)
    fx2 = func(x2)
    calls += 2
    
    prev_len = b - a
    
    while (b - a) > eps:
        iteration += 1
        current_len = b - a
        ratio = prev_len / current_len if iteration > 1 else 0
        
        if verbose:
            history.append({
                'i': iteration,
                'x1': x1, 'x2': x2,
                'fx1': fx1, 'fx2': fx2,
                'a': a, 'b': b,
                'len': current_len,
                'ratio': ratio
            })
            
        prev_len = current_len

        if fx1 < fx2:
            b = x2
            x2 = x1
            fx2 = fx1
            x1 = b - (b - a) / phi
            fx1 = func(x1)
            calls += 1
        else:
            a = x1
            x1 = x2
            fx1 = fx2
            x2 = a + (b - a) / phi
            fx2 = func(x2)
            calls += 1
            
    min_x = (a + b) / 2
    return min_x, func(min_x), calls, history

def method_fibonacci(func, a, b, eps, verbose=False):
    fibs = [1, 1]
    limit = (b - a) / eps
    while fibs[-1] < limit:
        fibs.append(fibs[-1] + fibs[-2])
    
    n = len(fibs) - 1 
    
    x1 = a + (fibs[n-2] / fibs[n]) * (b - a)
    x2 = a + (fibs[n-1] / fibs[n]) * (b - a)
    
    fx1 = func(x1)
    fx2 = func(x2)
    calls = 2
    
    history = []
    
    for k in range(1, n - 1):
        current_len = b - a
        
        if verbose:
            history.append({
                'i': k,
                'x1': x1, 'x2': x2,
                'fx1': fx1, 'fx2': fx2,
                'a': a, 'b': b,
                'len': current_len,
                'ratio': fibs[n-k+1]/fibs[n-k] if k < n else 0 # Примерное соотношение
            })

        if fx1 > fx2:
            a = x1
            x1 = x2
            fx1 = fx2
            x2 = a + (fibs[n - k - 1] / fibs[n - k]) * (b - a)
            fx2 = func(x2)
            calls += 1
        else:
            b = x2
            x2 = x1
            fx2 = fx1
            x1 = a + (fibs[n - k - 2] / fibs[n - k]) * (b - a)
            fx1 = func(x1)
            calls += 1
            
    delta = eps / 10
    x2 = x1 + delta
    fx2 = func(x2)
    calls += 1
    
    if fx1 > fx2:
        a = x1
    else:
        b = x2
        
    min_x = (a + b) / 2
    
    if verbose:
        history.append({
            'i': n - 1,
            'x1': x1, 'x2': x2,
            'fx1': fx1, 'fx2': fx2,
            'a': a, 'b': b,
            'len': b - a,
            'ratio': 1.0
        })
        
    return min_x, func(min_x), calls, history

def search_interval(func, x0, delta=0.1):
    history = []
    
    f0 = func(x0)
    
    if func(x0 + delta) < f0:
        x1 = x0 + delta
        h = delta
    elif func(x0 - delta) < f0:
        x1 = x0 - delta
        h = -delta
    else:
        return [x0 - delta, x0 + delta], history

    k = 1
    xk = x1
    xk_prev = x0
    history.append({'k': 0, 'x': x0, 'f': f0})
    history.append({'k': 1, 'x': x1, 'f': func(x1)})

    while True:
        h *= 2 
        xk_next = xk + h
        fk_next = func(xk_next)
        fk = func(xk)
        
        history.append({'k': k + 1, 'x': xk_next, 'f': fk_next})
        
        if not (fk < fk_next):
            interval = sorted([xk_prev, xk_next])
            return interval, history
        
        xk_prev = xk
        xk = xk_next
        k += 1

def print_table(history, method_name):
    print(f"\nТаблица результатов: {method_name}")
    print("-" * 105)
    print(f"{'i':<4} | {'x1':^12} | {'x2':^12} | {'f(x1)':^12} | {'f(x2)':^12} | {'a':^12} | {'b':^12} | {'b-a':^12} | {'ratio':^8}")

    for row in history:
        # ratio может быть 0 на первой итерации

    print("-" * 105)
