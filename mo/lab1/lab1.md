def method_dichotomy(func, a, b, eps):
    delta = eps / 2.1
    calls = 0
    history = []
    iteration = 0
    prev_len = b - a
    while (b - a) > eps:
        iteration += 1
        current_len = b - a
        ratio = prev_len / current_len if iteration > 1 else 0
        x1 = (a + b - delta) / 2
        x2 = (a + b + delta) / 2
        
        fx1 = func(x1)
        fx2 = func(x2)
        calls += 2  # Дихотомия: всегда 2 вызова за шаг
        
        history.append({
            'i': iteration,
            'x1': x1, 'x2': x2,
            'fx1': fx1, 'fx2': fx2,
            'a': a, 'b': b,
            'len': b - a,
            'ratio': ratio
        })
        prev_len = current_len
        if fx1 < fx2:
            b = x2
        else:
            a = x1
            
    min_x = (a + b) / 2
    return min_x, func(min_x), calls, history

            fx1 = fx2
            x2 = a + (b - a) / phi
            fx2 = func(x2)
            calls += 1
            
    min_x = (a + b) / 2
    return min_x, func(min_x), calls, history


def method_fibonacci(func, a, b, eps):
    fibs = [1, 1]
    limit = (b - a) / eps
    while fibs[-1] < limit:
        fibs.append(fibs[-1] + fibs[-2])
    n = len(fibs) - 1
    
    # Начальные точки (2 вызова)
    x1 = a + (fibs[n-2] / fibs[n]) * (b - a)
    x2 = a + (fibs[n-1] / fibs[n]) * (b - a)
    fx1 = func(x1)
    fx2 = func(x2)
    calls = 2
    
    history = []
    
    # Цикл идет n-2 раза
    for k in range(1, n - 1):
        history.append({
            'i': k,
            'x1': x1, 'x2': x2,
            'fx1': fx1, 'fx2': fx2,
            'a': a, 'b': b,
            'len': b - a,
            'ratio': fibs[n-k+1]/fibs[n-k] if k < n else 0 # Примерное соотношение
        })

        if fx1 > fx2:
            a = x1
            x1 = x2
            fx1 = fx2
            x2 = a + (fibs[n - k - 1] / fibs[n - k]) * (b - a)
            fx2 = func(x2)
            calls += 1
        else:
            b = x2
            x2 = x1
            fx2 = fx1
            x1 = a + (fibs[n - k - 2] / fibs[n - k]) * (b - a)
            fx1 = func(x1)
            calls += 1
            
    delta = eps / 100
    x2 = x1 + delta
    fx2 = func(x2)
    calls += 1
    
    history.append({
        'i': n - 1,
        'x1': x1, 'x2': x2,
        'fx1': fx1, 'fx2': fx2,
        'a': a, 'b': b,
        'len': b - a,
        'ratio': 1.0
    })

    if fx1 > fx2:
        a = x1
    else:
        b = x2
        
    min_x = (a + b) / 2
    return min_x, func(min_x), calls, history
